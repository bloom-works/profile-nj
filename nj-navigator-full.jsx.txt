// ─── HELPERS ──────────────────────────────────────────────────────
const Divider = () => <div style={{ height: 1, background: C.faint, margin: "0" }} />;

function Tag({ label, color = C.green, bg = C.greenLight }) {
  return (
    <span style={{ fontSize: 10, padding: "3px 9px", borderRadius: 20, background: bg, color, letterSpacing: "0.06em", fontWeight: "bold", whiteSpace: "nowrap", textTransform: "uppercase" }}>
      {label}
    </span>
  );
}

function Btn({ children, variant = "primary", onClick, style = {} }) {
  const base = { border: "none", borderRadius: 8, padding: "13px 20px", fontSize: 14, cursor: "pointer", fontFamily: "Georgia, serif", transition: "opacity 0.15s", ...style };
  const variants = {
    primary:   { background: C.navy,  color: C.white },
    gold:      { background: C.gold,  color: C.navy,  fontWeight: "bold" },
    ghost:     { background: C.white, color: C.steel, border: `1px solid ${C.border}` },
    ghostNavy: { background: "rgba(255,255,255,0.1)", color: C.white, border: "1px solid rgba(255,255,255,0.2)" },
  };
  return <button style={{ ...base, ...variants[variant] }} onClick={onClick} onMouseEnter={e => e.currentTarget.style.opacity = "0.85"} onMouseLeave={e => e.currentTarget.style.opacity = "1"}>{children}</button>;
}

function Input({ label, type = "text", placeholder }) {
  return (
    <div style={{ marginBottom: 16 }}>
      <label style={{ fontSize: 12, color: C.steel, display: "block", marginBottom: 6 }}>{label}</label>
      <input type={type} placeholder={placeholder} style={{ width: "100%", border: `1px solid ${C.border}`, borderRadius: 6, padding: "11px 14px", fontSize: 14, color: C.navy, fontFamily: "Georgia, serif", boxSizing: "border-box", outline: "none", background: C.sand }} />
    </div>
  );
}

// ─── LAYOUT SHELL ─────────────────────────────────────────────────

  const pct = step === 1 ? 8 : step === 2 ? 24 : step === 3 ? 44 : step === 4 ? 62 : step === 5 ? 82 : 100;
  const crumbs = ["Intake", "Questions", "Mapping", "Results", "Apply", "Dashboard"];
  const screenKeys = ["s1", "s2", "s3", "s4", "s5", "s6"];
  return (
    <div style={{ minHeight: "100vh", background: C.sand, fontFamily: "Georgia, serif", display: "flex", flexDirection: "column" }}>
      <style>{`
        @keyframes fadeUp { from { opacity:0; transform:translateY(14px); } to { opacity:1; transform:translateY(0); } }
        @keyframes checkPop { 0%{transform:scale(0);opacity:0} 70%{transform:scale(1.2)} 100%{transform:scale(1);opacity:1} }
        @keyframes pulse { 0%,100%{opacity:1} 50%{opacity:0.35} }
        @keyframes spin { to { transform:rotate(360deg); } }
        .fade { animation: fadeUp 0.38s ease both; }
        .hover-lift:hover { transform: translateY(-1px); box-shadow: 0 4px 16px rgba(10,34,64,0.1); }
        .crumb-link:hover { text-decoration: underline; }
      `}</style>

      {/* Header */}
      <div style={{ background: C.navy, padding: "14px 28px", display: "flex", alignItems: "center", justifyContent: "space-between", borderBottom: `3px solid ${C.gold}` }}>
        <div style={{ display: "flex", alignItems: "center", gap: 12 }}>
          <div style={{ width: 34, height: 34, background: C.gold, borderRadius: 4, display: "flex", alignItems: "center", justifyContent: "center", fontSize: 16, fontWeight: "bold", color: C.navy }}>NJ</div>
          <div>
            <div style={{ color: C.white, fontSize: 14, fontWeight: "bold", letterSpacing: "0.02em" }}>NJ Service Navigator</div>
            <div style={{ color: C.muted, fontSize: 10, letterSpacing: "0.07em", textTransform: "uppercase" }}>Pilot · NJ Innovation Authority</div>

        {crumbs.map((c, i) => {
          const isCurrent = i + 1 === step;
          const isVisited = i + 1 <= maxVisited && !isCurrent;
          const isClickable = isVisited && onNavigate;
          return (
            <span key={i} style={{ display: "flex", alignItems: "center", gap: 6 }}>
              {i > 0 && <span style={{ color: C.muted }}>›</span>}
              <span

// ─── STEP 2: CLARIFY ─────────────────────────────────────────────
function Step2({ category, onNext, onBack, maxVisited, onNavigate }) {
  const questions = clarifyMap[category] || clarifyMap.food;
  const [currentQ, setCurrentQ] = useState(0);
  const [answers, setAnswers] = useState({});
  const [selected, setSelected] = useState(null);

  const handleAnswer = (opt) => {
    setSelected(opt);
    setTimeout(() => {
      const newA = { ...answers, [questions[currentQ].id]: opt };
      setAnswers(newA);
      setSelected(null);
      if (currentQ < questions.length - 1) setCurrentQ(currentQ + 1);
      else onNext(newA);
    }, 280);
  };

  const cat = categories.find(c => c.id === category);

  return (
    <Shell step={2} maxVisited={maxVisited} onNavigate={onNavigate}>
      <div className="fade">
        <StepLabel n={2} total={5} title="A few quick questions" sub={`Question ${currentQ + 1} of ${questions.length} — helps us find the most relevant programs.`} />

        {/* Dots */}
        <div style={{ display: "flex", gap: 6, alignItems: "center", marginBottom: 24 }}>
          {questions.map((_, i) => (
            <div key={i} style={{ width: i === currentQ ? 22 : 7, height: 7, borderRadius: 4, background: i < currentQ ? C.green : i === currentQ ? C.gold : C.border, transition: "all 0.3s" }} />
          ))}
        </div>

        <div style={{ background: C.white, border: `1px solid ${C.border}`, borderRadius: 12, padding: "28px", boxShadow: "0 2px 12px rgba(10,34,64,0.05)", marginBottom: 16 }}>
          <div style={{ display: "inline-flex", alignItems: "center", gap: 6, background: C.goldLight, borderRadius: 20, padding: "4px 12px", marginBottom: 20 }}>
            <span style={{ fontSize: 14 }}>{cat?.icon}</span>
            <span style={{ fontSize: 10, color: "#8a6020", letterSpacing: "0.07em", textTransform: "uppercase" }}>{cat?.label}</span>
          </div>
          <div style={{ fontSize: 19, color: C.navy, marginBottom: 24, lineHeight: 1.5 }}>{questions[currentQ].text}</div>
          <div style={{ display: "flex", flexDirection: "column", gap: 9 }}>
            {questions[currentQ].options.map(opt => (
              <button
                key={opt}
                onClick={() => handleAnswer(opt)}
                style={{ background: selected === opt ? C.navy : C.sand, border: `1.5px solid ${selected === opt ? C.navy : C.border}`, borderRadius: 8, padding: "13px 18px", cursor: "pointer", textAlign: "left", fontSize: 14, color: selected === opt ? C.white : C.navy, transition: "all 0.15s", display: "flex", justifyContent: "space-between", alignItems: "center", fontFamily: "Georgia, serif" }}
                onMouseEnter={e => { if (selected !== opt) { e.currentTarget.style.borderColor = C.gold; e.currentTarget.style.background = C.goldLight; } }}
                onMouseLeave={e => { if (selected !== opt) { e.currentTarget.style.borderColor = C.border; e.currentTarget.style.background = C.sand; } }}
              >
                <span>{opt}</span>
                {selected === opt && <span>✓</span>}
              </button>
            ))}
          </div>
        </div>

        {Object.keys(answers).length > 0 && (
          <div style={{ background: C.greenLight, border: `1px solid #b8dcc8`, borderRadius: 8, padding: "12px 16px", marginBottom: 12 }}>
            <div style={{ fontSize: 10, color: C.green, textTransform: "uppercase", letterSpacing: "0.09em", marginBottom: 8 }}>Your answers so far</div>
            {Object.entries(answers).map(([key, val]) => {
              const q = questions.find(q => q.id === key);
              return (
                <div key={key} style={{ fontSize: 12, color: C.steel, display: "flex", justifyContent: "space-between", padding: "3px 0" }}>
                  <span>{q?.text}</span><span style={{ color: C.green, fontWeight: "bold", marginLeft: 12 }}>{val}</span>
                </div>
              );
            })}
          </div>
        )}

        <button onClick={onBack} style={{ background: "none", border: "none", color: C.muted, fontSize: 12, cursor: "pointer", padding: 0 }}>← Back</button>
      </div>
    </Shell>
  );
}

// ─── STEP 3: MAPPING ─────────────────────────────────────────────
function Step3({ category, onNext, maxVisited, onNavigate }) {
  const [activeStep, setActiveStep] = useState(0);
  const [completed, setCompleted] = useState([]);
  const [done, setDone] = useState(false);

  useEffect(() => {
    if (activeStep >= mappingLog.length) { setDone(true); return; }
    const t = setTimeout(() => {
      setCompleted(prev => [...prev, activeStep]);
      setActiveStep(s => s + 1);
    }, mappingLog[activeStep].dur);
    return () => clearTimeout(t);
  }, [activeStep]);

  const programs = resultsMap[category] || resultsMap.food;
  const pct = (completed.length / mappingLog.length) * 100;

  return (
    <Shell step={3} maxVisited={maxVisited} onNavigate={onNavigate}>
      <div className="fade">
        <StepLabel n={3} total={5} title={done ? "Search complete" : "Searching across NJ agencies…"} sub={done ? `We found ${programs.length} programs that may match your situation.` : "This usually takes less than 10 seconds."} />

        {/* Progress ring area */}
        <div style={{ background: C.navy, borderRadius: 12, padding: "22px", marginBottom: 20 }}>
          <div style={{ display: "flex", alignItems: "center", justifyContent: "space-between", marginBottom: 14 }}>
            <div style={{ fontSize: 11, color: C.muted, textTransform: "uppercase", letterSpacing: "0.09em" }}>Search log</div>
            <div style={{ fontSize: 12, color: done ? C.green : C.gold }}>{done ? `${programs.length} programs found` : `${completed.length} / ${mappingLog.length}`}</div>
          </div>
          <div style={{ background: "rgba(255,255,255,0.08)", borderRadius: 4, height: 4, marginBottom: 18 }}>
            <div style={{ height: 4, background: done ? C.green : C.gold, width: `${pct}%`, transition: "width 0.4s ease", borderRadius: 4 }} />
          </div>
          <div style={{ display: "flex", flexDirection: "column", gap: 8 }}>
            {mappingLog.map((s, i) => {
              const isComplete = completed.includes(i);
              const isActive = activeStep === i;
              const isPending = activeStep < i;
              return (
                <div key={i} style={{ display: "flex", alignItems: "center", gap: 10, opacity: isPending ? 0.25 : 1, transition: "opacity 0.3s" }}>
                  <div style={{ width: 20, height: 20, borderRadius: "50%", flexShrink: 0, background: isComplete ? C.green : isActive ? C.gold : "transparent", border: `2px solid ${isComplete ? C.green : isActive ? C.gold : C.steel}`, display: "flex", alignItems: "center", justifyContent: "center", fontSize: 10, color: C.white, transition: "all 0.3s", animation: isActive ? "pulse 0.7s ease infinite" : "none" }}>
                    {isComplete ? "✓" : ""}
                  </div>
                  <div style={{ flex: 1 }}>
                    <div style={{ fontSize: 12, color: isComplete ? C.white : isActive ? C.gold : C.steel, transition: "color 0.3s" }}>{s.icon} {s.label}</div>
                    {(isActive || isComplete) && <div style={{ fontSize: 10, color: C.muted, marginTop: 1 }}>{s.detail}</div>}
                  </div>
                  {isActive && <div style={{ fontSize: 10, color: C.gold, animation: "pulse 1s step-end infinite" }}>running</div>}
                </div>
              );
            })}
          </div>

// ─── ROOT ─────────────────────────────────────────────────────────
export default function App() {
  const [screen, setScreen]       = useState("s1");
  const [category, setCategory]   = useState(null);
  const [answers, setAnswers]     = useState({});
  const [selectedProgram, setSelectedProgram] = useState(null);
  const [maxVisited, setMaxVisited] = useState(1);

  const screenToStep = { s1: 1, s2: 2, s3: 3, s4: 4, s5: 5, s6: 6 };

  const goTo = (key) => {
    setScreen(key);
  };

  const advance = (key) => {
    const step = screenToStep[key];
    setMaxVisited(prev => Math.max(prev, step));
    setScreen(key);
  };

  const programs = category ? (resultsMap[category] || resultsMap.food) : [];

  const restart = () => { setScreen("s1"); setCategory(null); setAnswers({}); setSelectedProgram(null); setMaxVisited(1); };

  const navProps = { maxVisited, onNavigate: goTo };

  if (screen === "s1") return <Step1 {...navProps} onNext={cat => { setCategory(cat); advance("s2"); }} />;
  if (screen === "s2") return <Step2 {...navProps} category={category} onNext={ans => { setAnswers(ans); advance("s3"); }} onBack={() => goTo("s1")} />;
  if (screen === "s3") return <Step3 {...navProps} category={category} onNext={() => advance("s4")} />;
  if (screen === "s4") return <Step4 {...navProps} category={category} onApply={p => { setSelectedProgram(p); advance("s5"); }} onBack={() => goTo("s2")} />;
  if (screen === "s5") return <Step5 {...navProps} program={selectedProgram} allPrograms={programs} onRestart={restart} onNext={() => advance("s6")} />;
  if (screen === "s6") return <Step6 {...navProps} onRestart={restart} />;
  return null;
}
